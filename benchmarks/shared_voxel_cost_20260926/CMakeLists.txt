cmake_minimum_required(VERSION 3.16)
project(shared_voxel_cost)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
get_filename_component(WORKSPACE_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT GENERATED_SRC)
  set(GENERATED_SRC "${WORKSPACE_SRC}/artifacts/shared_voxel_cost_20260926/generated")
endif()
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(gng_cpu REQUIRED)
find_package(voxel_idx REQUIRED)
set(ROS_DEPS rclcpp rclcpp_components sensor_msgs std_msgs geometry_msgs std_srvs
    tf2 tf2_ros tf2_eigen visualization_msgs ais_gng_msgs voxel_msgs nlohmann_json)
foreach(dep IN LISTS ROS_DEPS)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(shared_voxel_cost main.cpp "${GENERATED_SRC}/fvg.cpp")
target_include_directories(shared_voxel_cost PRIVATE
  "${GENERATED_SRC}" "${CMAKE_CURRENT_SOURCE_DIR}"
  "${WORKSPACE_SRC}/fuzzy_voxel_grid/include"
  "${WORKSPACE_SRC}/gng_vlut_system/src" "${WORKSPACE_SRC}/gng_vlut_system/src/core")
target_link_libraries(shared_voxel_cost voxel_idx::point_cloud_store gng_cpu::voxel_framework Eigen3::Eigen OpenMP::OpenMP_CXX)
ament_target_dependencies(shared_voxel_cost ${ROS_DEPS})
include(CheckIPOSupported)
check_ipo_supported(RESULT has_ipo)
if(has_ipo)
  set_property(TARGET shared_voxel_cost PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
