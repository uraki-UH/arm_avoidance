cmake_minimum_required(VERSION 3.16)
project(gng_spatial_tree LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "ビルド構成")
option(GNG_DETERMINISTIC_BENCHMARK "比較専用の乱数・時間刻み固定" ON)
set(SPATIAL_TREE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../SpatialTree/include"
  CACHE PATH "SpatialTreeヘッダーの場所")
set(bsp3d_include_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../bsp3d/include"
  CACHE PATH "bsp3dヘッダーの場所")
find_package(Boost REQUIRED)
find_package(OpenSSL REQUIRED)
file(GLOB_RECURSE core_sources CONFIGURE_DEPENDS src/*.cpp)
list(FILTER core_sources EXCLUDE REGEX "cugng(_grid_reference)?\\.cpp$")
enable_testing()
foreach(variant grid spatial bsp3d)
  if(variant STREQUAL "grid")
    set(search_source src/cpu/cugng_grid_reference.cpp)
  else()
    set(search_source src/cpu/cugng.cpp)
  endif()
  add_library(gng_${variant} SHARED ${core_sources} ${search_source})
  target_include_directories(gng_${variant} PUBLIC include PRIVATE ${Boost_INCLUDE_DIRS})
  target_compile_options(gng_${variant} PRIVATE -O3 -DNDEBUG -fvisibility=hidden)
  target_compile_definitions(gng_${variant} PRIVATE GNG_VERSION=0)
  if(GNG_DETERMINISTIC_BENCHMARK)
    target_compile_definitions(gng_${variant} PRIVATE GNG_DETERMINISTIC_BENCHMARK=1)
  endif()
  if(variant STREQUAL "spatial")
    target_include_directories(gng_${variant} PRIVATE "${SPATIAL_TREE_INCLUDE_DIR}")
    target_compile_definitions(gng_${variant} PRIVATE GNG_USE_SPATIAL_TREE=1)
  elseif(variant STREQUAL "bsp3d")
    target_include_directories(gng_${variant} PRIVATE "${bsp3d_include_dir}")
    target_compile_definitions(gng_${variant} PRIVATE GNG_USE_SPATIAL_TREE=1 GNG_USE_BSP3D=1)
  endif()
  target_link_libraries(gng_${variant} PRIVATE OpenSSL::SSL OpenSSL::Crypto ykpiv)
  target_link_options(gng_${variant} PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/gng_cpu.v")
  foreach(test_name gng_parameter_update_api_test gng_spatial_coverage_api_test
      gng_priority_input_api_test gng_neighbors_api_test gng_observation_api_test
      gng_training_event_api_test gng_map_delta_api_test gng_static_age_api_test)
    add_executable(${test_name}_${variant} test/${test_name}.cpp)
    target_link_libraries(${test_name}_${variant} PRIVATE gng_${variant})
    if(test_name STREQUAL "gng_static_age_api_test")
      add_test(NAME ${test_name}_${variant} COMMAND ${test_name}_${variant} 100)
    else()
      add_test(NAME ${test_name}_${variant} COMMAND ${test_name}_${variant})
    endif()
  endforeach()
endforeach()

# 内部索引の全走査照合。共有ライブラリの非公開シンボルに依存しない直接リンク。
foreach(variant spatial bsp3d)
  add_executable(gng_${variant}_nearest_test test/gng_spatial_nearest_test.cpp
    src/cpu/cugng.cpp src/utils/node.cpp src/utils/param.cpp
    src/utils/vec3f.cpp src/utils/utils.cpp)
  target_include_directories(gng_${variant}_nearest_test PRIVATE include)
  target_compile_definitions(gng_${variant}_nearest_test PRIVATE GNG_USE_SPATIAL_TREE=1 GNG_VERSION=0)
  if(variant STREQUAL "spatial")
    target_include_directories(gng_${variant}_nearest_test PRIVATE "${SPATIAL_TREE_INCLUDE_DIR}")
  else()
    target_include_directories(gng_${variant}_nearest_test PRIVATE "${bsp3d_include_dir}")
    target_compile_definitions(gng_${variant}_nearest_test PRIVATE GNG_USE_BSP3D=1)
  endif()
  add_test(NAME gng_${variant}_nearest_test COMMAND gng_${variant}_nearest_test)
endforeach()

# bsp3d同梱の動的索引検証。最適化を維持し、検証用assertのみ有効化。
add_executable(bsp3d_index_test "${bsp3d_include_dir}/../examples/test_bsp3d.cpp")
target_include_directories(bsp3d_index_test PRIVATE "${bsp3d_include_dir}")
target_compile_options(bsp3d_index_test PRIVATE -UNDEBUG)
add_test(NAME bsp3d_index_test COMMAND bsp3d_index_test)
