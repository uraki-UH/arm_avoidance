cmake_minimum_required(VERSION 3.16)
project(gng_minimal_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "ビルド構成")
option(GNG_DETERMINISTIC_BENCHMARK "比較専用の乱数・時間刻み固定" ON)
set(bsp3d_include_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../bsp3d/include" CACHE PATH "bsp3dヘッダーの場所")
find_package(Boost REQUIRED)
find_package(OpenSSL REQUIRED)
file(GLOB_RECURSE core_sources CONFIGURE_DEPENDS src/*.cpp)
enable_testing()
foreach(variant tree_raw tree_voxel grid_raw grid_voxel)
  set(target gng_minimal_${variant})
  add_library(${target} SHARED ${core_sources})
  target_include_directories(${target} PUBLIC include PRIVATE ${Boost_INCLUDE_DIRS} "${bsp3d_include_dir}")
  target_compile_options(${target} PRIVATE -O3 -DNDEBUG -fvisibility=hidden)
  target_compile_definitions(${target} PRIVATE GNG_VERSION=0)
  if(GNG_DETERMINISTIC_BENCHMARK)
    target_compile_definitions(${target} PRIVATE GNG_DETERMINISTIC_BENCHMARK=1)
  endif()
  set(variant_definitions "")
  if(variant MATCHES "^grid")
    list(APPEND variant_definitions GNG_USE_NODE_GRID=1)
  endif()
  if(variant MATCHES "voxel$")
    list(APPEND variant_definitions GNG_USE_INPUT_VOXELS=1)
  endif()
  target_compile_definitions(${target} PRIVATE ${variant_definitions})
  target_link_libraries(${target} PRIVATE OpenSSL::SSL OpenSSL::Crypto ykpiv)
  target_link_options(${target} PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/gng_cpu.v")
  foreach(test_name gng_neighbors_api_test gng_training_event_api_test gng_map_delta_api_test minimal_input_api_test)
    add_executable(${variant}_${test_name} test/${test_name}.cpp)
    target_compile_definitions(${variant}_${test_name} PRIVATE ${variant_definitions})
    target_link_libraries(${variant}_${test_name} PRIVATE ${target})
    add_test(NAME ${variant}_${test_name} COMMAND ${variant}_${test_name})
  endforeach()
endforeach()
foreach(search tree grid)
  if(search STREQUAL "tree")
    set(test_source test/gng_spatial_nearest_test.cpp)
  else()
    set(test_source test/node_grid_test.cpp)
  endif()
  add_executable(${search}_nearest_test ${test_source} src/cpu/cugng.cpp src/cpu/node_grid.cpp
    src/utils/node.cpp src/utils/param.cpp src/utils/vec3f.cpp src/utils/utils.cpp)
  target_include_directories(${search}_nearest_test PRIVATE include "${bsp3d_include_dir}")
  target_compile_definitions(${search}_nearest_test PRIVATE GNG_VERSION=0)
  if(search STREQUAL "grid")
    target_compile_definitions(${search}_nearest_test PRIVATE GNG_USE_NODE_GRID=1)
  endif()
  add_test(NAME ${search}_nearest_test COMMAND ${search}_nearest_test)
endforeach()
add_executable(input_voxels_test test/input_voxels_test.cpp src/utils/vec3f.cpp)
target_include_directories(input_voxels_test PRIVATE include)
add_test(NAME input_voxels_test COMMAND input_voxels_test)
