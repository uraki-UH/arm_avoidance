cmake_minimum_required(VERSION 3.16)
project(gng_bsp3d_minimal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "ビルド構成")
option(GNG_DETERMINISTIC_BENCHMARK "比較専用の乱数・時間刻み固定" ON)
set(bsp3d_include_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../bsp3d/include" CACHE PATH "bsp3dヘッダーの場所")
find_package(Boost REQUIRED)
find_package(OpenSSL REQUIRED)
file(GLOB_RECURSE core_sources CONFIGURE_DEPENDS src/*.cpp)
add_library(gng_bsp3d_minimal SHARED ${core_sources})
target_include_directories(gng_bsp3d_minimal PUBLIC include PRIVATE ${Boost_INCLUDE_DIRS} "${bsp3d_include_dir}")
target_compile_options(gng_bsp3d_minimal PRIVATE -O3 -DNDEBUG -fvisibility=hidden)
target_compile_definitions(gng_bsp3d_minimal PRIVATE GNG_VERSION=0)
if(GNG_DETERMINISTIC_BENCHMARK)
  target_compile_definitions(gng_bsp3d_minimal PRIVATE GNG_DETERMINISTIC_BENCHMARK=1)
endif()
target_link_libraries(gng_bsp3d_minimal PRIVATE OpenSSL::SSL OpenSSL::Crypto ykpiv)
target_link_options(gng_bsp3d_minimal PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/gng_cpu.v")
enable_testing()
foreach(test_name gng_neighbors_api_test gng_training_event_api_test gng_map_delta_api_test minimal_input_api_test)
  add_executable(${test_name} test/${test_name}.cpp)
  target_link_libraries(${test_name} PRIVATE gng_bsp3d_minimal)
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
add_executable(gng_bsp3d_nearest_test test/gng_spatial_nearest_test.cpp
  src/cpu/cugng.cpp src/utils/node.cpp src/utils/param.cpp src/utils/vec3f.cpp src/utils/utils.cpp)
target_include_directories(gng_bsp3d_nearest_test PRIVATE include "${bsp3d_include_dir}")
target_compile_definitions(gng_bsp3d_nearest_test PRIVATE GNG_VERSION=0)
add_test(NAME gng_bsp3d_nearest_test COMMAND gng_bsp3d_nearest_test)
